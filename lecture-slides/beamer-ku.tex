% The PDF version of the decks (format: beamer) in the look of ku_clean.scss:
% Roboto, dark grey text, KU red titles, slide number and logo at the bottom.
\usepackage[sfdefault,light]{roboto}
\usepackage[scale=0.85]{roboto-mono}
\definecolor{kured}{HTML}{901A1E}
\definecolor{jet}{HTML}{3D3D3D}
\definecolor{kugrey}{HTML}{666666}

\setbeamercolor{normal text}{fg=jet}
\setbeamercolor{structure}{fg=kured}
\setbeamercolor{frametitle}{fg=kured}
\setbeamercolor{title}{fg=kured}
\setbeamercolor{subtitle}{fg=kugrey}
\setbeamercolor{section title}{fg=kured}
\setbeamercolor{alerted text}{fg=kured}
\setbeamercolor{block title}{fg=kured}
\setbeamerfont{frametitle}{size=\Large}
\setbeamerfont{title}{size=\LARGE}
\setbeamerfont{subtitle}{size=\large,shape=\itshape}
\setbeamerfont{section title}{size=\huge}
\setbeamertemplate{itemize items}{\textbullet}
\setbeamertemplate{caption}{\insertcaption}
\hypersetup{colorlinks=true, urlcolor=kured, linkcolor=kured}

\setbeamertemplate{frametitle}{\vspace{0.6em}\usebeamerfont{frametitle}\insertframetitle\par\vspace{0.2em}}
\setbeamertemplate{footline}{%
  \hspace{1.2em}\raisebox{0.5em}{\includegraphics[height=0.55cm]{ku_logo.png}}%
  \hfill\raisebox{0.8em}{\color{kugrey}\scriptsize\insertframenumber}\hspace{1.2em}}
\setbeamertemplate{title page}{%
  \vfill
  {\usebeamerfont{title}\usebeamercolor[fg]{title}\inserttitle\par}
  \ifx\insertsubtitle\empty\else\vspace{0.6em}
  {\usebeamerfont{subtitle}\usebeamercolor[fg]{subtitle}\insertsubtitle\par}\fi
  \vspace{2em}
  {\insertauthor\par}
  \vspace{0.3em}
  {\small\color{kugrey}\insertinstitute\par}
  \vspace{0.3em}
  {\small\color{kugrey}\insertdate\par}
  \vfill}
\setbeamertemplate{section page}{%
  \vfill{\usebeamerfont{section title}\usebeamercolor[fg]{section title}\insertsection\par}\vfill}
% Code a bit smaller than the text
\AtBeginEnvironment{Shaded}{\footnotesize}
